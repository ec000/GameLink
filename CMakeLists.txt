cmake_minimum_required(VERSION 3.21)
project(GameLink VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GAMELINK_BUILD_GUI "Build Qt 6 desktop GUI" ON)
add_library(gamelink_core client/core/p2p.cpp)
target_include_directories(gamelink_core PUBLIC client/core)
if(MSVC)
 target_compile_options(gamelink_core PRIVATE /W4 /utf-8)
else()
 target_compile_options(gamelink_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(gamelink_core_tests tests/core_tests.cpp)
 target_link_libraries(gamelink_core_tests PRIVATE gamelink_core)
 add_test(NAME core COMMAND gamelink_core_tests)
endif()
if(GAMELINK_BUILD_GUI)
 # The Windows installer and ZIP use a flat directory. Qt's deployment
 # helper must put DLLs/qt.conf beside the executable, not in default bin/.
 if(WIN32)
  set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Flat Windows application layout" FORCE)
 endif()
 find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
 qt_standard_project_setup()
 add_library(gamelink_client STATIC client/gui/transport.cpp client/gui/window.cpp client/gui/window_ui.cpp client/gui/window_installer.cpp client/platform/zerotier_environment.cpp client/platform/zerotier_installer.cpp)
 target_include_directories(gamelink_client PUBLIC client/gui client/platform)
 if(WIN32)
  target_link_libraries(gamelink_client PUBLIC advapi32 msi shell32 wintrust crypt32)
 endif()
 target_link_libraries(gamelink_client PUBLIC gamelink_core Qt6::Widgets Qt6::Network)
 target_compile_definitions(gamelink_client PUBLIC QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII GAMELINK_VERSION="${PROJECT_VERSION}")
 qt_add_executable(GameLink client/gui/main.cpp)
 target_link_libraries(GameLink PRIVATE gamelink_client)
 if(MSVC)
  target_compile_options(gamelink_client PRIVATE /utf-8 /W4)
 else()
  target_compile_options(gamelink_client PRIVATE -Wall -Wextra -Wpedantic)
 endif()
 if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  qt_add_executable(gamelink_qt_tests tests/qt_tests.cpp)
  set_target_properties(gamelink_qt_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(gamelink_qt_tests PRIVATE gamelink_client Qt6::Test)
  add_dependencies(gamelink_qt_tests GameLink)
  if(MSVC)
   target_compile_options(gamelink_qt_tests PRIVATE /utf-8)
  endif()
  add_test(NAME qt_client COMMAND gamelink_qt_tests -o qt-client-results.txt,txt -o -,txt)
  # The generic offscreen plugin has no Windows font database. Use the
  # native platform there so screenshots validate real text rendering.
  if(WIN32)
   set(gamelink_test_platform windows)
  else()
   set(gamelink_test_platform offscreen)
  endif()
  set_tests_properties(qt_client PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${gamelink_test_platform}" TIMEOUT 90)
  add_test(NAME gui_startup COMMAND GameLink --self-test)
  set_tests_properties(gui_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${gamelink_test_platform}" TIMEOUT 15)
 endif()
 if(WIN32)
  set_target_properties(GameLink PROPERTIES WIN32_EXECUTABLE TRUE)
  target_sources(GameLink PRIVATE client/platform/windows/app.rc)
  target_include_directories(GameLink PRIVATE client/platform/windows)
  if(MSVC)
   target_link_options(GameLink PRIVATE /MANIFEST:NO)
  endif()
  target_link_libraries(GameLink PRIVATE iphlpapi)
 endif()
 if(MSVC)
  target_compile_options(GameLink PRIVATE /utf-8 /W4)
 endif()
 add_custom_command(TARGET GameLink POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  ${CMAKE_SOURCE_DIR}/profiles $<TARGET_FILE_DIR:GameLink>/profiles)
 install(TARGETS GameLink RUNTIME DESTINATION . BUNDLE DESTINATION .)
 install(DIRECTORY profiles DESTINATION .)
 qt_generate_deploy_app_script(TARGET GameLink OUTPUT_SCRIPT qt_deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
 if(WIN32)
  # Qt may capture install directories while loading Qt6Core. Override the
  # documented deployment variable in the install process as well.
  install(CODE "set(QT_DEPLOY_BIN_DIR \".\")")
 endif()
 install(SCRIPT ${qt_deploy_script})
 set(CPACK_GENERATOR ZIP)
 set(CPACK_PACKAGE_FILE_NAME "GameLink-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}")
 include(CPack)
endif()
